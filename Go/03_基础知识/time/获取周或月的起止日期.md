# 获取周或月的起止日期

```go
// 基于周期类型，获取当前日期所属周期的起始日期和结束日期
// @return startAt,endAt,error
func getPeriodRange(periodType string) {
	// 先获取当前周期类别，然后基于当前日期，获取周期的起始和结束日期
	nowTime := time.Now().AddDate(0, 0, -3)
	fmt.Println("nowTime:", nowTime.Format("2006-01-02"))
	var periodStart time.Time
	var periodEnd time.Time
	switch periodType {
	case "week":
		// 计算今天是周几，time.Monday代表周一，time.Sunday代表周日
		weekday := nowTime.Weekday()

		if weekday == time.Sunday {
			// 如果今天是周天
			periodStart = nowTime.AddDate(0, 0, -6)
			periodEnd = nowTime
		} else if weekday == time.Monday {
			// 如果今天是周一
			periodStart = nowTime
			periodEnd = nowTime.AddDate(0, 0, 6)
		} else {
			offsetDays := int(weekday - time.Monday)
			periodStart = nowTime.AddDate(0, 0, -offsetDays).Truncate(24 * time.Hour)
			periodEnd = periodStart.AddDate(0, 0, 6)
		}

	case "month":
		log.Default().Println("Month:", nowTime.Month())
		periodStart = time.Date(nowTime.Year(), nowTime.Month(), 1, 0, 0, 0, 0, time.Local)
		periodEnd = periodStart.AddDate(0, 1, -1)
	default:

	}

	fmt.Printf("周期第一天:%s,最后一天：%s \n", periodStart.Format("2006-01-02"), periodEnd.Format("2006-01-02"))
}
```
