# 01-注意事项


以下面的代码为例，

* `CpTitleLayout` 必须放在 `CoordinatorLayout` 中
    * 这样才能确保既能让它一直固定在顶部，又能确保 `CoordinatorLayout-AppBarLayout` 内部的 `TabLayout` 在滚动时悬浮在屏幕顶部。
    * 如果把 `CpTitleLayout` 也一起放到了 `CoordinatorLayout` 内部，那么整个 `AppBarLayout` 都将无法滚动。
* 如果直接把 `ViewPage2` 作为  `CoordinatorLayout` 的子 View, 那么 `ViewPager2` 中 Fragment 中必须包含 `RecyclerView` 或 `NestedScrollView`, 否则滚动事件无法传递。

```xml
<?xml version="1.0" encoding="utf-8"?>

<layout
    xmlns:android="http://schemas.android.com/apk/res/android"
    xmlns:app="http://schemas.android.com/apk/res-auto"
    xmlns:tools="http://schemas.android.com/tools">

    <data>

        <import type="cn.jnpolice.course.R" />
        <import type="android.view.View" />
        <import type="android.text.TextUtils"/>

        <variable
            name="vm"
            type="cn.jnpolice.course.pages.course.detail.vm.CourseDetailVm" />

    </data>

    <LinearLayout
        android:layout_width="match_parent"
        android:layout_height="match_parent"
        android:orientation="vertical">

        <!--这个必须放在 CoordinatorLayout 外面，不然会导致整个列表无法滚动 -->
        <cn.jnpolice.course.util.components.titlelayout.CpTitleLayout
            android:id="@+id/title_layout"
            android:layout_width="match_parent"
            android:layout_height="wrap_content"
            app:layout_scrollFlags="scroll"
            app:title='@{vm.detail.name}' />

        <androidx.coordinatorlayout.widget.CoordinatorLayout
            android:layout_width="match_parent"
            android:layout_height="match_parent">

            <com.google.android.material.appbar.AppBarLayout
                android:layout_width="match_parent"
                android:layout_height="wrap_content">

                <androidx.constraintlayout.widget.ConstraintLayout
                    android:id="@+id/ll_summary"
                    android:layout_width="match_parent"
                    android:layout_height="wrap_content"
                    android:layout_marginBottom="@dimen/dp15"
                    android:background="@drawable/shape_rect_rounded_btm5radius_solid_white"
                    android:orientation="vertical"
                    android:paddingLeft="15dp"
                    android:paddingTop="10dp"
                    android:paddingRight="15dp"
                    android:paddingBottom="10dp"
                    android:visibility="visible"
                    app:layout_scrollFlags="scroll">

                    <ImageView
                        android:id="@+id/iv_cover"
                        android:layout_width="match_parent"
                        android:layout_height="0dp"
                        android:scaleType="fitXY"
                        app:imgSrc="@{vm.detail.coverImg}"
                        app:layout_constraintDimensionRatio="1:0.57"
                        app:layout_constraintLeft_toLeftOf="parent"
                        app:layout_constraintRight_toRightOf="parent"
                        app:layout_constraintTop_toTopOf="parent"
                        app:placeholder="@{R.drawable.ic_course_def}"
                        app:radius="@{8}"
                        tools:src="@color/primary" />

                    <TextView
                        android:id="@+id/tv_readed_count"
                        android:layout_width="wrap_content"
                        android:layout_height="wrap_content"
                        android:layout_marginTop="15dp"
                        android:layout_marginBottom="10dp"
                        android:background="@drawable/shape_rect_rounded_5radius_solid_white"
                        android:backgroundTint="#F4F4F7"
                        android:paddingLeft="10dp"
                        android:paddingTop="4dp"
                        android:paddingRight="10dp"
                        android:paddingBottom="4dp"
                        android:text='@{"浏览量："+vm.totalReadCount}'
                        android:textColor="#8F98B4"
                        android:textSize="12sp"
                        app:layout_constraintLeft_toLeftOf="parent"
                        app:layout_constraintTop_toBottomOf="@id/iv_cover"
                        tools:text="浏览量：199" />

                    <!-- 收藏-->
                    <LinearLayout
                        android:id="@+id/ll_star"
                        android:layout_width="wrap_content"
                        android:layout_height="wrap_content"
                        android:background="@drawable/shape_rect_with_50r"
                        android:backgroundTint="@{vm.starred ? @color/bg_tint_starred:@color/bg_tint_un_star}"
                        android:gravity="center"
                        android:onClick="@{vm::onStarClick}"
                        android:paddingLeft="10dp"
                        android:paddingTop="4dp"
                        android:paddingRight="10dp"
                        android:paddingBottom="4dp"
                        app:layout_constraintBottom_toBottomOf="@id/tv_readed_count"
                        app:layout_constraintRight_toRightOf="parent"
                        app:layout_constraintTop_toTopOf="@id/tv_readed_count">

                        <ImageView
                            android:layout_width="15dp"
                            android:layout_height="15dp"
                            android:src="@{vm.starred ? @drawable/ic_yi_shou_cang:@drawable/ic_star_disable}"
                            tools:src="@drawable/ic_star_disable" />

                        <TextView
                            android:layout_width="wrap_content"
                            android:layout_height="wrap_content"
                            android:layout_marginLeft="5dp"
                            android:text='@{vm.starred ?"已收藏":"收藏"}'
                            android:textColor="@{vm.starred?@color/starred:@color/text_2nd}"
                            android:textSize="13sp"
                            tools:text="收藏" />
                    </LinearLayout>

                    <TextView
                        android:id="@+id/tv_name"
                        android:layout_width="match_parent"
                        android:layout_height="wrap_content"
                        android:layout_marginTop="10dp"
                        android:text="@{vm.detail.name}"
                        android:textColor="#333333"
                        android:textSize="14sp"
                        android:textStyle="bold"
                        app:layout_constraintLeft_toLeftOf="parent"
                        app:layout_constraintRight_toRightOf="parent"
                        app:layout_constraintTop_toBottomOf="@id/tv_readed_count"
                        tools:text="这是课程名称" />

                    <cn.jnpolice.course.util.components.textview.CpExpandableTextView
                        android:id="@+id/tv_desc"
                        android:layout_width="match_parent"
                        android:layout_height="wrap_content"
                        android:layout_marginTop="10dp"
                        android:textColor="#999999"
                        android:textSize="14sp"
                        android:maxLines="2"
                        android:visibility="@{TextUtils.isEmpty(vm.detail.summary)?View.GONE:View.VISIBLE}"
                        android:text="@{vm.detail.summary}"
                        app:layout_constraintBottom_toBottomOf="parent"
                        app:layout_constraintLeft_toLeftOf="parent"
                        app:layout_constraintRight_toRightOf="parent"
                        app:layout_constraintTop_toBottomOf="@id/tv_name"
                        tools:text="无论你是想解决工作中 SpringBoot 整合的疑难问题，还是想掌握性能优化、监控运维、项目部署的" />
                </androidx.constraintlayout.widget.ConstraintLayout>

                <com.google.android.material.tabs.TabLayout
                    android:id="@+id/tabLayout"
                    android:layout_width="match_parent"
                    android:layout_height="wrap_content"
                    android:background="@drawable/bg_course_detail_tab"
                    android:visibility="visible"
                    app:tabRippleColor="@null" />
            </com.google.android.material.appbar.AppBarLayout>

                <androidx.viewpager2.widget.ViewPager2
                    android:id="@+id/view_pager"
                    android:layout_width="match_parent"
                    android:layout_height="match_parent"
                    android:visibility="visible"
                    tools:background="#DCD" 
                    app:layout_behavior="@string/appbar_scrolling_view_behavior"
                    />
        </androidx.coordinatorlayout.widget.CoordinatorLayout>
    </LinearLayout>
</layout>
```
